    function route() {
      const h = location.hash;
      document.getElementById('friendsView').hidden = true;
      document.getElementById('communityView').hidden = true;
      document.getElementById('adminView').hidden = !h.startsWith('#/admin');
      if (h.startsWith('#/u/')) {
        const [username, dashboardSection] = h.slice(4).split('#', 2);
        showProfile(decodeURIComponent(username), false, dashboardSection || null);
      } else if (h === '#/inbox') {
        showInbox();
      } else if (h === '#/friends') {
        showFriends();
      } else if (h === '#/community' || h.startsWith('#/community/')) {
        const id = h.startsWith('#/community/') ? Number(h.slice('#/community/'.length)) : null;
        if (id && Number.isInteger(id)) showCommunityDetail(id); else showCommunities();
      } else if (h === '#/admin') {
        showAdmin();
      } else if (h === '#/search') {
        showUserSearch();
      } else if (h === '#/settings' || h.startsWith('#/settings#')) {
        showSettings(h.split('#')[2] || null);
      } else if (h === '#/signup') {
        showSignup();
      } else {
        showHome();
      }
    }
