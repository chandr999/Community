
    function hideSocialViews() {
      ['homeView', 'profileView', 'inboxView', 'friendsView', 'communityView', 'userSearchView', 'settingsView', 'signupView', 'adminView']
        .forEach(id => { const node = document.getElementById(id); if (node) node.hidden = true; });
      document.getElementById('appPausedView').hidden = true;
    }

    async function showFriends() {
      stopInboxPolling(); hideSocialViews();
      const view = document.getElementById('friendsView'); view.hidden = false; view.replaceChildren();
      view.append(el('h1', 'Friends', 'inbox-title'));
      if (!loggedIn) { view.append(el('p', 'Log in to see the people you follow.', 'empty')); return; }
      const response = await fetch('/api/me/friends');
      if (!response.ok) { view.append(el('p', 'Friends could not be loaded. Please try again.', 'empty')); return; }
      const people = await response.json();
      const following = el('section'); following.append(el('h2', `Following Â· ${people.length}`));
      if (!people.length) following.append(el('p', 'People you follow will appear here.', 'empty'));
      const grid = el('div', undefined, 'social-grid');
      people.forEach(person => {
        const card = el('article', undefined, 'social-card');
        const row = el('div', undefined, 'social-person'); row.append(avatarEl(person.avatar_url), profileLink(person.username, '@' + person.username)); card.append(row);
        const actions = el('div', undefined, 'social-card-actions');
        const favorite = el('button', person.is_favorite ? 'â˜… Favorited' : 'â˜† Add to favorites');
        favorite.onclick = async () => { const r = await fetch('/api/users/' + encodeURIComponent(person.username) + '/favorite', { method: person.is_favorite ? 'DELETE' : 'POST' }); if (r.ok) showFriends(); };
        const unfollow = el('button', 'Unfollow');
        unfollow.onclick = async () => { if (window.confirm(`Unfollow @${person.username}?`)) { const r = await fetch('/api/users/' + encodeURIComponent(person.username) + '/follow', { method: 'DELETE' }); if (r.ok) showFriends(); } };
        actions.append(favorite, unfollow); card.append(actions); grid.append(card);
      });
      following.append(grid); view.append(following);
      const favoriteSection = el('section'); favoriteSection.append(el('h2', 'Favorites'));
      const favorites = people.filter(p => p.is_favorite);
      if (!favorites.length) favoriteSection.append(el('p', 'Mark people as favorites for quick access.', 'empty'));
      const favGrid = el('div', undefined, 'social-grid');
      favorites.forEach(person => { const card = el('article', undefined, 'social-card'); const row = el('div', undefined, 'social-person'); row.append(avatarEl(person.avatar_url), profileLink(person.username, '@' + person.username)); card.append(row); favGrid.append(card); });
      favoriteSection.append(favGrid); view.append(favoriteSection);
    }

    function communitySideItem(group, activeId, openCommunity) {
      const button = el('button', undefined, 'community-side-item' + (Number(activeId) === group.id ? ' active' : ''));
      button.type = 'button';
      const mark = group.image_url ? el('img') : el('span', group.name.slice(0, 1).toUpperCase(), 'community-side-mark');
      if (group.image_url) { mark.src = group.image_url; mark.alt = ''; }
      const copy = el('span', undefined, 'community-side-copy');
      copy.append(el('strong', group.name), el('span', group.is_disabled ? 'Access paused' : `${group.member_count} members`));
      button.append(mark, copy);
      button.disabled = group.is_disabled;
      button.onclick = () => openCommunity(group);
      return button;
    }
