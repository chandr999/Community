      async function loadConversations() {
        const response = await fetch('/api/conversations');
        if (!response.ok) {
          conversationList.replaceChildren(el('p', 'Could not load your conversations.', 'activity-empty'));
          return;
        }
        cachedConversations = await response.json();
        updateChatMenuActions();
        conversationList.textContent = '';
        if (cachedConversations.length === 0) {
          conversationList.append(el('p', 'Your conversations will appear here.', 'activity-empty'));
          activeConversationId = null;
          headingTitle.textContent = 'Choose a conversation';
          messageList.replaceChildren(el('p', 'Start a chat to say hello.', 'message-empty'));
          messageInput.disabled = false; sendButton.disabled = false;
          return;
        }
        if (!cachedConversations.some(conversation => conversation.id === activeConversationId)) {
          activeConversationId = cachedConversations[0].id;
        }
        renderConversationItems();
        if (activeConversationId !== null) {
          const active = cachedConversations.find(conversation => conversation.id === activeConversationId);
          if (active) headingTitle.textContent = active.title ||
            active.members.filter(name => name !== currentUsername).join(', ') || 'Conversation';
          const muted = Boolean(active?.is_muted);
          messageInput.disabled = muted;
          sendButton.disabled = muted;
          emojiButton.disabled = muted; gifButton.disabled = muted; stickerButton.disabled = muted; plusButton.disabled = muted;
          messageHint.textContent = muted ? 'This chat has been muted by an administrator. You can still view its messages.' : 'Photos, GIFs, and videos up to 50 MB. Messages refresh every 5 seconds.';
        }
        updateChatMenuActions();
      }

      function renderConversationItems() {
        conversationList.textContent = '';
        const visibleConversations = cachedConversations.filter(conversation => {
          if (inboxFilter === 'unread') return conversation.unread_count > 0;
          if (inboxFilter === 'groups') return conversation.is_group;
          if (inboxFilter === 'community') return conversation.is_group && /community/i.test(conversation.title || '');
          return true;
        });
        if (visibleConversations.length === 0) {
          conversationList.append(el('p', inboxFilter === 'unread' ? 'No unread messages.' :
            inboxFilter === 'community' ? 'No community chats yet.' : 'No chats in this view.', 'activity-empty'));
          return;
        }
        visibleConversations.forEach(conversation => {
          const item = el('button', undefined, 'conversation-item' +
            (conversation.id === activeConversationId ? ' active' : ''));
          const avatar = el('div', conversation.is_group ? 'G' : (conversation.members.find(name => name !== currentUsername) || '?')[0].toUpperCase(), 'activity-icon');
          const copy = el('span', undefined, 'conversation-copy');
          const title = conversation.title ||
            conversation.members.filter(name => name !== currentUsername).join(', ');
          const name = el('span', title || 'Conversation', 'conversation-name');
          if (conversation.is_muted) name.append(el('span', ' Â· Muted', 'conversation-unread'));
          if (conversation.unread_count > 0) name.append(el('span', String(conversation.unread_count), 'conversation-unread'));
          copy.append(name, el('span', conversation.latest_message || 'No messages yet', 'conversation-preview'));
          item.append(avatar, copy);
          item.onclick = async () => {
            activeConversationId = conversation.id;
            chat.hidden = false;
            chat.classList.add('chat-window');
            await loadMessages(conversation.id);
            await loadConversations();
          };
          conversationList.append(item);
        });
      }

      async function loadMessages(conversationId) {
        if (conversationId === null || conversationId !== activeConversationId) return;
        const response = await fetch('/api/conversations/' + conversationId + '/messages');
        if (!response.ok) {
          messageList.replaceChildren(el('p', 'Could not load messages for this conversation.', 'message-empty'));
          return;
        }
        const messages = await response.json();
        if (conversationId !== activeConversationId) return;
        messageList.textContent = '';
        if (messages.length === 0) {
          messageList.append(el('p', 'No messages yet. Say hello!', 'message-empty'));
          return;
        }
        messages.forEach(message => {
          const bubble = el('div', undefined, 'message-bubble' +
            (message.sender === currentUsername ? ' mine' : ''));
          if (message.sender !== currentUsername) {
            bubble.append(el('span', message.sender, 'message-sender'));
          }
          if (message.body) bubble.append(el('div', message.body));
          if (message.image_url) {
            if (message.media_type === 'video') {
              const video = el('video');
              video.src = message.image_url;
              video.controls = true;
              video.playsInline = true;
              video.className = 'message-media';
              bubble.append(video);
            } else {
              const image = el('img');
              image.src = message.image_url;
              image.alt = 'Media shared by ' + message.sender;
              image.className = 'message-media';
              bubble.append(image);
            }
          }
          bubble.append(el('span', new Date(message.created_at).toLocaleString(), 'message-time'));
          if (message.can_delete) {
            const remove = el('button', 'Delete', 'message-delete');
            remove.type = 'button';
            remove.title = 'Delete this message';
            remove.onclick = async () => {
              if (!confirm('Delete this message?')) return;
              const result = await fetch('/api/conversations/' + conversationId + '/messages/' + message.id, { method: 'DELETE' });
              if (!result.ok) { alert('Could not delete this message.'); return; }
              await loadConversations();
              await loadMessages(conversationId);
            };
            bubble.append(remove);
          }
          messageList.append(bubble);
        });
        messageList.scrollTop = messageList.scrollHeight;
      }
    }
